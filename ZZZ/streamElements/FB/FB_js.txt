// ⚠️ 填入你的粉絲專頁 ID（Numeric Page ID）
const PAGE_ID = '451422554732276'; 

function autoFetchFBViewers() {
  if (!PAGE_ACCESS_TOKEN || PAGE_ACCESS_TOKEN.includes('你的FB')) {
    document.getElementById('fb-count').innerText = '未設定 Token';
    return;
  }

  // 加入 limit=1 限制回傳數量，避免舊直播干擾
  const searchUrl = `https://graph.facebook.com/v18.0/${PAGE_ID}/live_videos?status=LIVE&limit=1&access_token=${PAGE_ACCESS_TOKEN}`;

  fetch(searchUrl)
    .then(res => {
      if (!res.ok) throw new Error(`HTTP Error: ${res.status}`);
      return res.json();
    })
    .then(data => {
      if (data && data.data && data.data.length > 0) {
        // 取得最新的直播 ID
        const liveVideoId = data.data[0].id;
        
        // 查詢該直播的線上人數
        return fetch(`https://graph.facebook.com/v18.0/${liveVideoId}?fields=live_views&access_token=${PAGE_ACCESS_TOKEN}`);
      } else {
        // 目前沒有開播
        throw new Error('OFFLINE');
      }
    })
    .then(res => {
      if (!res) return null;
      if (!res.ok) throw new Error(`HTTP Error: ${res.status}`);
      return res.json();
    })
    .then(liveData => {
      if (liveData && typeof liveData.live_views !== 'undefined') {
        document.getElementById('fb-count').innerText = liveData.live_views;
      }
    })
    .catch(err => {
      if (err.message === 'OFFLINE') {
        document.getElementById('fb-count').innerText = '0 (未開播)';
      } else {
        console.error('FB API 讀取失敗:', err);
        // API 發生錯誤時（如 Rate Limit），維持舊數值不直接歸 0，避免 Overlay 閃爍
      }
    });
}

// 首次載入執行，並建議調整為 30 秒刷新一次以防 Rate Limit
autoFetchFBViewers();
setInterval(autoFetchFBViewers, 30000);
